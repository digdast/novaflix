Docker Compose Deployment
One-command container deployment for Unraid, TrueNAS SCALE, Synology, or Linux.
services:
  novaflix:
    image: ghcr.io/digdast/novaflix:latest
    container_name: novaflix
    restart: unless-stopped
    ports:
      - "8096:8096"
      - "8097:8097"
    volumes:
      - ./data:/app/data
      - /path/to/movies:/media/movies:ro
      - /path/to/tv:/media/tv:ro
    environment:
      - NOVAFLIX_PORT=8096
      - NOVAFLIX_REMOTE_ENABLED=true
